# BUILD LIBRARY FROM LOCAL FiLES FOR THE MAIN LIBRARY
file(GLOB SRC "*.hpp" "*.cpp") 
include_directories($ENV{JPACPLOTTER}/src)

add_library(JPACPLOTTER SHARED ${SRC} )

        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} )
